Run it yourself

It needs Linux on x86-64, Nix with flakes, jq and python3.

git clone https://github.com/logos-messaging/chat-sim && cd chat-sim && git checkout 1200dfb93a6e839378b69902a66b8def2df9a115
pinned=$(jq -r '.nodes."logos-chat".locked.rev' flake.lock)
sed -i "s/$pinned/8b9cd8efa7f413be073c222b76746e7add669e17/g" flake.nix Cargo.toml
nix flake lock
nix develop -c cargo build --release -p harness --bin harness-run
python3 tools/sim sweep tools/sim/scenarios/grow-waku-loss-1.json --from 4 --runs 1 --one-process target/release/harness-run --out tmp/grow-waku-loss-1-seed-3
