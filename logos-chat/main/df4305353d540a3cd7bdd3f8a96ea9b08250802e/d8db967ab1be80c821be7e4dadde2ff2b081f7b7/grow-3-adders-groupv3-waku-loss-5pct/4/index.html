Run it yourself

It needs Linux on x86-64, Nix with flakes, jq and python3.

git clone https://github.com/logos-messaging/chat-sim && cd chat-sim && git checkout d8db967ab1be80c821be7e4dadde2ff2b081f7b7
pinned=$(jq -r '.nodes."logos-chat".locked.rev' flake.lock)
sed -i "s/$pinned/df4305353d540a3cd7bdd3f8a96ea9b08250802e/g" flake.nix Cargo.toml
nix flake lock
nix develop -c cargo build --release -p harness --bin harness-run
python3 tools/sim sweep tools/sim/scenarios/grow-3-adders-groupv3-waku-loss-5pct.json --from 4 --runs 1 --one-process target/release/harness-run --out tmp/grow-3-adders-groupv3-waku-loss-5pct-seed-3
