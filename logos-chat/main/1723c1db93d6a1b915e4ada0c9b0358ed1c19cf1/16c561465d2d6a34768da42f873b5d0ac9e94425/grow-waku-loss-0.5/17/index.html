Run it yourself

It needs Linux on x86-64, Nix with flakes, jq and python3.

git clone https://github.com/logos-messaging/chat-sim && cd chat-sim && git checkout 7a5b74c386c96b07decddc0917c96efc0c547458
pinned=$(jq -r '.nodes."logos-chat".locked.rev' flake.lock)
sed -i "s/$pinned/1723c1db93d6a1b915e4ada0c9b0358ed1c19cf1/g" flake.nix Cargo.toml
nix flake lock
nix develop -c cargo build --release -p harness --bin harness-run
python3 tools/sim sweep tools/sim/scenarios/grow-waku-loss-0.5.json --from 17 --runs 1 --one-process target/release/harness-run --out tmp/grow-waku-loss-0.5-seed-16
