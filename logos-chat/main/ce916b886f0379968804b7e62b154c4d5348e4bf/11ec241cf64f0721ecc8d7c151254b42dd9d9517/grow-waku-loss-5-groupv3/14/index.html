Run it yourself

It needs Linux on x86-64, Nix with flakes, jq and python3.

git clone https://github.com/logos-messaging/chat-sim && cd chat-sim && git checkout 11ec241cf64f0721ecc8d7c151254b42dd9d9517
pinned=$(jq -r '.nodes."logos-chat".locked.rev' flake.lock)
sed -i "s/$pinned/ce916b886f0379968804b7e62b154c4d5348e4bf/g" flake.nix Cargo.toml
nix flake lock
nix develop -c cargo build --release -p harness --bin harness-run
python3 tools/sim sweep tools/sim/scenarios/grow-waku-loss-5-groupv3.json --from 14 --runs 1 --one-process target/release/harness-run --out tmp/grow-waku-loss-5-groupv3-seed-13
